        function switchView(viewName) {

            ['schedule', 'projects', 'grading'].forEach(v => {

                document.getElementById(`view-${v}`).classList.add('hidden');

                document.getElementById(`tab-${v}`).classList.replace('tab-active', 'tab-inactive');

            document.getElementById(`tab-${viewName}`).classList.replace('tab-inactive', 'tab-active');

        }

        function renderTimeline() {

            const container = document.getElementById('syllabus-timeline');

            courseData.modules.forEach(mod => {

                const modEl = document.createElement('div');

                modEl.className = "mb-6";

                modEl.innerHTML = `<h3 class="text-xs font-bold text-slate-400 uppercase tracking-wider mb-2 px-2">Module ${mod.id}: ${mod.title}</h3>`;

                mod.weeks.forEach(week => {

                    const btn = document.createElement('button');

                    btn.className = "w-full text-left p-3 rounded-lg mb-2 transition-all duration-200 inactive-week group border border-slate-100";

                    btn.innerHTML = `<div class="flex justify-between items-center mb-1"><span class="text-xs font-bold text-uit-primary bg-uit-light px-2 py-0.5 rounded">Week ${week.num}</span></div><h4 class="font-semibold text-slate-700 text-sm group-hover:text-uit-primary">${week.title}</h4>`;

                    btn.onclick = () => showWeekDetail(week, mod.title, btn);

                    modEl.appendChild(btn);

                });

                container.appendChild(modEl);

            });

        }

            btn.classList.remove('inactive-week'); btn.classList.add('active-week');

            const panel = document.getElementById('detail-panel');

            panel.style.opacity = '0';

            setTimeout(() => {

                panel.innerHTML = `

        function getDomainStyle(domain) {

            const styles = {

                'SDN': 'bg-blue-100 text-blue-700 border-blue-200',

                'Cloud': 'bg-purple-100 text-purple-700 border-purple-200',

                'Edge': 'bg-orange-100 text-orange-700 border-orange-200',

                'Security': 'bg-red-100 text-red-700 border-red-200',

                'IoT': 'bg-emerald-100 text-emerald-700 border-emerald-200',

                'Multimedia': 'bg-pink-100 text-pink-700 border-pink-200'

            };

            return styles[domain] || 'bg-slate-100 text-slate-700 border-slate-200';

        }


        function filterProjects(category, btnElement) {

            // Update UI

            if (btnElement) {

                document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));

                btnElement.classList.add('active');

            }

            renderProjects(category);

        }


        function renderProjects(category) {

            const container = document.getElementById('project-grid');

            container.innerHTML = ''; // Clear existing

            

            const filtered = category === 'All' 

                ? courseData.projects 

                : courseData.projects.filter(p => p.domain.includes(category) || (category === 'IoT' && p.domain === 'Multimedia')); // Include multimedia in IoT/Wireless for simplicity or keep separate


            filtered.forEach(p => {

                const domainStyle = getDomainStyle(p.domain);

                const card = document.createElement('div');

                card.className = "bg-white p-6 rounded-xl border border-slate-200 shadow-sm hover:shadow-md transition group flex flex-col h-full animate-fade-in";

                card.innerHTML = `

                    <div class="flex justify-between items-start mb-4">

                        <span class="${domainStyle} text-xs font-bold px-2 py-1 rounded border">${p.domain}</span>

                    </div>

                    <h3 class="font-bold text-slate-900 text-lg mb-2 leading-tight">${p.title}</h3>

                    <p class="text-slate-600 text-sm mb-4 leading-relaxed flex-grow">${p.desc}</p>

                    <div class="flex flex-wrap gap-2 mt-auto pt-4 border-t border-slate-50">

                        ${p.tools.map(t => `<span class="text-xs bg-slate-50 text-slate-500 px-2 py-1 rounded border border-slate-100">${t}</span>`).join('')}

                    </div>

                container.appendChild(card);
